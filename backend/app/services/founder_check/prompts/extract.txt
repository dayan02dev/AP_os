{
  "name": string,
  "education": [{"degree": string, "field": string, "institution": string, "year": string}],
  "roles": [{"title": string, "org": string, "start": string, "end": string, "ownership": string, "impact": string}],
  "ventures": [{"name": string, "role": string, "stage": string, "outcome": string}],
  "patents": [string],
  "publications": [string],
  "grants": [string],
  "deployments_pilots": [string],
  "open_source": [string],
  "skills": [string],
  "raw_highlights": [string]
}

"ownership" = what the person themselves built/led (be specific about scope; distinguish individual contribution from team credit). If the résumé is empty or unreadable, return {}.
